cmake_minimum_required(VERSION 3.16)
project(octool_cli LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "octool-cli requires the Linux x86_64 hardware backend")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
include(CTest)
add_subdirectory(../gui/core core)
file(STRINGS ../VERSION OCTOOL_VERSION LIMIT_COUNT 1)
add_library(octool_linux STATIC ../gui/platform/linux_cpu.cpp ../gui/platform/linux_hwio.cpp
    ../gui/platform/linux_inventory_native.cpp ../port/hal/octool_hwio.c)
target_include_directories(octool_linux PUBLIC ../gui)
target_link_libraries(octool_linux PUBLIC octool_core)
add_library(octool_commands STATIC commands.cpp json.cpp)
target_link_libraries(octool_commands PUBLIC octool_linux)
target_compile_definitions(octool_commands PRIVATE OCTOOL_VERSION="${OCTOOL_VERSION}")
add_executable(octool-cli main.cpp)
target_link_libraries(octool-cli PRIVATE octool_commands)
foreach(target octool_linux octool_commands octool-cli)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
if(BUILD_TESTING)
    add_executable(cli_test tests.cpp)
    target_link_libraries(cli_test PRIVATE octool_commands)
    target_compile_options(cli_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME cli_commands COMMAND cli_test)
    set_tests_properties(cli_commands PROPERTIES TIMEOUT 60)
    find_package(Python3 3.8 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli_process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_process.py" "$<TARGET_FILE:octool-cli>" "$<TARGET_FILE:cli_test>")
    set_tests_properties(cli_process PROPERTIES TIMEOUT 60)
endif()
install(TARGETS octool-cli RUNTIME DESTINATION bin)
install(FILES ../docs/headless-cli.md ../docs/umc-offline.md ../docs/intel-uncore-recovery.md ../docs/ddr5-xmp-recovery.md ../docs/ddr5-expo-recovery.md DESTINATION share/doc/octool-cli)
install(FILES ../gui/LICENSE DESTINATION share/licenses/octool-cli)
